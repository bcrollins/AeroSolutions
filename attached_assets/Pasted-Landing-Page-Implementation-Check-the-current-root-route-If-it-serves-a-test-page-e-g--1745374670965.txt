In server.js, ensure / serves the landing page:

app.get('/', (req, res) => {
  res.sendFile(path.join(__dirname, 'public', 'index.html'));
});



Move existing login or test routes to /login or /test:

app.get('/login', (req, res) => {
  res.sendFile(path.join(__dirname, 'public', 'login.html')); // Adjust as needed
});
app.get('/test', (req, res) => {
  res.send('Test page');
});
